local function roleOf(plr)
    local c, b = plr.Character, plr:FindFirstChildOfClass("Backpack")
    if (c and c:FindFirstChild("Knife")) or (b and b:FindFirstChild("Knife")) then return "Murderer" end
    if (c and c:FindFirstChild("Gun")) or (b and b:FindFirstChild("Gun")) then return "Sheriff" end
    return "Innocent"
end

local function newEspObj()
    local o = {
        shadow = {}, corners = {},
        pill     = newDraw("Square", {Filled = true, Color = C.pill}),
        pillLine = newDraw("Square", {Filled = true}),
        box      = newDraw("Square", {Filled = false, Thickness = 1}),

        dist     = newDraw("Text", {Size = 11, Center = true, Outline = true, Color = C.white}),
        tracer   = newDraw("Line", {Thickness = 1}),
    }
    for i = 1, 8 do
        o.shadow[i]  = newDraw("Line", {Thickness = 5})
        o.corners[i] = newDraw("Line", {Thickness = 1.5})
    end

local function setVis(o, v)
    for _, d in pairs(o) do
        if type(d) == "table" then
            for _, s in ipairs(d) do s.Visible = v end
        else
            d.Visible = v
        end
    end
end

local function removeEsp(plr)
    local o = esp[plr]
    if not o then return end
    for _, d in pairs(o) do
        if type(d) == "table" then
            for _, s in ipairs(d) do s:Remove() end
        else
            d:Remove()
        end
    end
    esp[plr] = nil
end

local function setSkin(char, on)
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then
            if on then
                if matOrig[p] == nil then matOrig[p] = p.Material end
                p.Material = Enum.Material.ForceField
            elseif matOrig[p] ~= nil then
                p.Material = matOrig[p]
                matOrig[p] = nil
            end
        end
    end
end

local chamsOn = {}
local function updateChams(plr, col, on)
    local char = plr.Character
    local h = chams[plr]
    if not (on and char) then
        if h then h:Destroy(); chams[plr] = nil end
        if char and chamsOn[plr] then setSkin(char, false); chamsOn[plr] = nil end
        return
    end
    if not h or h.Parent ~= char then
        if h then h:Destroy() end
        h = new("Highlight", char, {
            DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
            FillTransparency = 0.55, OutlineTransparency = 0,
        })
        chams[plr] = h
        setSkin(char, true)
        chamsOn[plr] = true
    end
    h.FillColor, h.OutlineColor = col, col
    h.OutlineTransparency = 0.05 + (math.sin(os.clock() * 4) + 1) * 0.1
end

local function drawCorners(o, x, y, w, h, col, alpha, on)
    local L = math.max(5, math.min(w, h) * 0.24)
    local segs = {
        {x, y, x + L, y}, {x, y, x, y + L},
        {x + w, y, x + w - L, y}, {x + w, y, x + w, y + L},
        {x, y + h, x + L, y + h}, {x, y + h, x, y + h - L},
        {x + w, y + h, x + w - L, y + h}, {x + w, y + h, x + w, y + h - L},
    }
    for i = 1, 8 do
        local g, c, sg = o.shadow[i], o.corners[i], segs[i]
        local a, b = Vector2.new(sg[1], sg[2]), Vector2.new(sg[3], sg[4])
        g.From, g.To, g.Color, g.Transparency, g.Visible = a, b, col, alpha * 0.35, on
        c.From, c.To, c.Color, c.Transparency, c.Visible = a, b, col, alpha, on
    end
end

local function updateEsp(plr, cam, vp, role, col)
    local o = esp[plr] or newEspObj()
    esp[plr] = o
    local char = plr.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    local head = char and char:FindFirstChild("Head")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    if not CFG.ESP or not hrp or not head or not hum or hum.Health <= 0 then
        setVis(o, false); return
    end

    local dist = (cam.CFrame.Position - hrp.Position).Magnitude
    if dist > CFG.MaxDist then setVis(o, false); return end
    local alpha = CFG.FadeDist and math.clamp(1 - (dist / CFG.MaxDist) * 0.6, 0.4, 1) or 1

    local topP, topOn = screenOf(cam, head.Position + Vector3.new(0, 0.7, 0))
    local botP, botOn = screenOf(cam, hrp.Position - Vector3.new(0, 3.1, 0))
    if not (topOn and botOn) then setVis(o, false); return end

    local h = botP.Y - topP.Y
    if h < 4 then setVis(o, false); return end
    local w  = h * 0.55
    local cx = screenOf(cam, hrp.Position).X
    local minX, minY = cx - w / 2, topP.Y
    local maxY = botP.Y

    if CFG.CornerBox then
        drawCorners(o, minX, minY, w, h, col, alpha, CFG.Box)
        o.box.Visible = false
    else
        drawCorners(o, minX, minY, w, h, col, alpha, false)
        o.box.Position, o.box.Size, o.box.Color = Vector2.new(minX, minY), Vector2.new(w, h), col
        o.box.Transparency = alpha
        o.box.Visible = CFG.Box
    end

    local pillW = #plr.Name * 7 + 18
    local pillX, pillY = cx - pillW / 2, minY - 24
    o.pill.Position, o.pill.Size = Vector2.new(pillX, pillY), Vector2.new(pillW, 16)
    o.pill.Transparency = alpha * 0.8
    o.pill.Visible = CFG.Name
    o.pillLine.Position, o.pillLine.Size = Vector2.new(pillX, pillY), Vector2.new(2, 16)
    o.pillLine.Color, o.pillLine.Transparency = col, alpha
    o.pillLine.Visible = CFG.Name

    local pct = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
    o.hpBack.Position, o.hpBack.Size = Vector2.new(minX - 6, minY - 1), Vector2.new(3, h + 2)
    o.hpBack.Transparency = alpha
    o.hpBack.Visible = CFG.HealthBar
    o.hp.Position = Vector2.new(minX - 5, maxY - h * pct)
    o.hp.Size = Vector2.new(1, h * pct)
    o.hp.Color = Color3.fromHSV(pct * 0.33, 1, 1)
    o.hp.Transparency = alpha
    o.hp.Visible = CFG.HealthBar

    o.role.Text = string.upper(role)
    o.role.Position, o.role.Color, o.role.Transparency = Vector2.new(cx, minY - 36), col, alpha
    o.role.Visible = CFG.Role

    o.name.Text = plr.Name
    o.name.Position, o.name.Color, o.name.Transparency = Vector2.new(cx, pillY + 1), C.text, alpha
    o.name.Visible = CFG.Name

    o.dist.Text = math.floor(dist) .. "m"
    o.dist.Position, o.dist.Transparency = Vector2.new(cx, maxY + 3), alpha
    o.dist.Visible = CFG.Distance

    local rp = screenOf(cam, hrp.Position)
    local from = CFG.TracerFrom == "center" and Vector2.new(vp.X / 2, vp.Y / 2) or Vector2.new(vp.X / 2, vp.Y)
    o.tracer.From, o.tracer.To = from, rp
    o.tracer.Color, o.tracer.Transparency = col, alpha
    o.tracer.Visible = CFG.Tracer
end

local pingCache, pingAt = 0.08, 0
local function getPing()
    local now = os.clock()
    if now - pingAt < 0.5 then return pingCache end
    pingAt = now
    local ok, v = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    pingCache = ok and (v / 1000) or 0.08
    return pingCache
end

local function predictPos(part)
    local t = 0
    if CFG.AimPred then
        if CFG.AimPing then t = t + getPing() end
        t = t + CFG.PredMs / 1000
    end
    return part.Position + part.AssemblyLinearVelocity * t
end

local function findTarget(cam, vp)
    local char0 = LP.Character
    if not char0 then return nil end
    local center = Vector2.new(vp.X / 2, vp.Y / 2)
    local bestD, bestPos, bestPlr = CFG.AimFov, nil, nil
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {char0}
    local wbActive = CFG.WallBang and roleOf(LP) == "Sheriff"

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            local char = plr.Character
            local hum  = char:FindFirstChildOfClass("Humanoid")
            local part = char:FindFirstChild(CFG.AimBone) or char:FindFirstChild("HumanoidRootPart")
            local role = roleOf(plr)
            local skip = (wbActive and role ~= "Murderer") or (CFG.AimMurderOnly and role ~= "Murderer")
            if hum and hum.Health > 0 and part and not skip then
                local aimPos = predictPos(part)
                local sp, on = screenOf(cam, aimPos)
                local d = (sp - center).Magnitude
                if on and d < bestD then
                    local ignoreWall = wbActive and role == "Murderer"
                    local visible = true
                    if CFG.AimVis and not ignoreWall then
                        local origin = cam.CFrame.Position
                        local r = workspace:Raycast(origin, aimPos - origin, params)
                        visible = (r == nil) or r.Instance:IsDescendantOf(char)
                    end
                    if visible then bestD, bestPos, bestPlr = d, aimPos, plr end
                end
            end
        end
    end
    return bestPos, bestPlr
end

local function wbFire(target)
    if not (CFG.WallBang and target) then return end
    local char = LP.Character
    local gun  = char and char:FindFirstChild("Gun")
    if not gun then return end
    if os.clock() - S.lastFire < 0.12 then return end
    S.lastFire = os.clock()
    pcall(function() gun:Activate() end)
end

-- ===== silent aim (sheriff): fires the gun remote at the murderer, camera untouched =====
-- short velocity history per player, averaged so a single spike doesn't throw the lead
local silVel = {}
RunService.Heartbeat:Connect(function()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP then
            local r = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local h = silVel[plr] or {}
                silVel[plr] = h
                table.insert(h, 1, r.AssemblyLinearVelocity)
                while #h > 5 do table.remove(h) end
            end
        end
    end
end)

-- lead the target by (ping + a fixed frame budget) along its averaged horizontal velocity
-- lead point: where the murderer will be after (ping if "add ping to lead" + extra ms).
-- uses the average horizontal velocity of the last 5 frames. vertical is ignored on purpose (jumps throw it off).
local function leadPoint(plr, root)
    local v = root.AssemblyLinearVelocity
    local h = silVel[plr]
    if h and #h > 0 then
        local s = Vector3.zero
        for _, vel in ipairs(h) do s = s + vel end
        v = s / #h
    end
    local t = 0
    if CFG.AimPred then
        if CFG.AimPing then t = t + getPing() end
        t = t + CFG.PredMs / 1000
    end
    return root.Position + Vector3.new(v.X * t, 0, v.Z * t)
end

-- silent: lead point + "silent forward push" along the murderer's move direction
local function silentAimPos(plr, root, hum)
    return leadPoint(plr, root) + hum.MoveDirection * CFG.SilentOffset
end

local function silentTarget()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character and roleOf(plr) == "Murderer" then
            local c    = plr.Character
            local hum  = c:FindFirstChildOfClass("Humanoid")
            local root = c:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and root then
                return plr, root, hum
            end
        end
    end
end

local function silentFire()
    if not CFG.SilentAim or not S.alive then return end
    if os.clock() - S.lastSilent < 0.15 then return end
    if roleOf(LP) ~= "Sheriff" then return end
    task.spawn(function()
        local char = LP.Character
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if not (char and hum and hum.Health > 0) then return end

        local plr, root, thum = silentTarget()
        if not plr then return end

        local gun = char:FindFirstChild("Gun")
        if not gun then
            local bp = LP:FindFirstChildOfClass("Backpack")
            local g  = bp and bp:FindFirstChild("Gun")
            if not g then return end
            hum:EquipTool(g)
            task.wait(0.1)
            gun = char:FindFirstChild("Gun")
            if not gun then return end
        end

        local shoot = gun:FindFirstChild("Shoot")
            or (gun:FindFirstChild("Events") and gun.Events:FindFirstChild("Shoot"))
        if not shoot then return end

        local hand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or char:FindFirstChild("HumanoidRootPart")
        local pos  = silentAimPos(plr, root, thum)

        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {char}
        local hit = workspace:Raycast(hand.Position, pos - hand.Position, params)
        if hit and not hit.Instance:IsDescendantOf(plr.Character) then return end

        S.lastSilent = os.clock()
        pcall(function()
            shoot:FireServer(CFrame.new(hand.Position), CFrame.new(pos))
        end)
    end)
end

local function weldModelTo(m, base)
    for _, p in ipairs(m:GetDescendants()) do
        if p:IsA("BasePart") then
            p.Anchored = false
            p.CanCollide = false
            p.Massless = true
            if p ~= base then new("WeldConstraint", p, {Part0 = base, Part1 = p}) end
        end
    end
end

local function clearAura()
    if S.aura then S.aura:Destroy(); S.aura = nil end
end

local function updateAura(char)

    if key == "Off" or not ASSET_AURAS[key] then clearAura(); return end
    local torso = char and (char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso"))
    if not torso then return end
    if S.aura and S.auraKey == key and S.aura.Parent == char then return end
    if S.auraKey == key and S.auraBusy then return end
    clearAura()

local function nearestPlayer()
    local c = LP.Character
    local r = c and c:FindFirstChild("HumanoidRootPart")
    if not r then return nil end
    local best, bestD = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        local tr = plr ~= LP and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
        if tr then
            local dd = (tr.Position - r.Position).Magnitude
            if dd < bestD then best, bestD = plr, dd end
        end
    end
    return best
end

local function resolveFling()
    if CFG.FlingName ~= "" then
        local q = string.lower(CFG.FlingName)
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and (string.find(string.lower(plr.Name), q, 1, true)
                or string.find(string.lower(plr.DisplayName), q, 1, true)) then
                return plr
            end
        end
        return nil
    end
    if CFG.FlingMode == "Nearest" then return nearestPlayer() end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and roleOf(plr) == CFG.FlingMode then return plr end
    end
    return nil
end

local function wbShoot()
    local char     = LP.Character
    local root     = char and char:FindFirstChild("HumanoidRootPart")
    local backpack = LP:FindFirstChildOfClass("Backpack")
    local gun      = char and (char:FindFirstChild("Gun") or (backpack and backpack:FindFirstChild("Gun")))
    if not (root and gun) then return false end

    local murderer
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and roleOf(plr) == "Murderer" then murderer = plr; break end
    end
    if not murderer then return false end

    local pos = wbPredict(murderer)
    if not pos then return false end

    if gun.Parent == backpack then
        gun.Parent = char
        task.wait()
    end

    local shoot = gun:FindFirstChild("Shoot")
    if shoot and shoot:IsA("RemoteEvent") then
        task.spawn(function()
            pcall(function() shoot:FireServer(CFrame.new(pos), CFrame.new(root.Position)) end)
        end)
        return true
    end

    local kl   = gun:FindFirstChild("KnifeLocal")
    local cb   = kl and kl:FindFirstChild("CreateBeam")
    local rf   = cb and cb:FindFirstChild("RemoteFunction")
    if rf then
        task.spawn(function()
            pcall(function() rf:InvokeServer(1, pos, "AH2") end)
        end)
        return true
    end
    return false
end

local function nlzShowList(quiet)
    if not httpReq then if not quiet then toast("executor has no http request") end return end

        if onlineLbl then onlineLbl.Text = "not set up: NLZ_URL is still the placeholder" end
        if not quiet then toast("nlz url not set") end
        return
    end
    task.spawn(function()
        local ok, res = pcall(function()
            return httpReq({Url = NLZ_URL .. "/online", Method = "GET"})
        end)
        local ok2, data = false, nil
        if ok and res and res.Body then
            ok2, data = pcall(function() return HttpService:JSONDecode(res.Body) end)
        end
        if not (ok2 and data) then
            if onlineLbl then onlineLbl.Text = "list failed" end
            if not quiet then toast("list failed") end
            return
        end
        local names = data.names or {}
        table.sort(names, function(x, y) return string.lower(x) < string.lower(y) end)
        if onlineLbl then
            onlineLbl.Text = (#names == 0) and "nobody online in the list" or table.concat(names, "\n")
        end
        if not quiet then toast(("nlz online: %d"):format(data.count or #names)) end
        for _, n in ipairs(names) do print("[nlz] " .. n) end
    end)
end

local charRecs = {}
local selfHidden = nil

local function setSelfHidden(char, hide)
    if not char then return end
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") and p.Name ~= "HumanoidRootPart" then
            p.LocalTransparencyModifier = hide and 1 or 0
        end
    end
end

local function dropCharModel(plr)
    local rec = charRecs[plr]
    if rec then
        if rec.model then rec.model:Destroy() end
        charRecs[plr] = nil
    end
end

local function ensureCharModel(plr, char, id)
    local rec = charRecs[plr]
    if rec and rec.char == char and rec.model and rec.model.Parent then return end
    dropCharModel(plr)
    charRecs[plr] = {char = char}
    loadTpl(id, function(tpl)
        local cur = charRecs[plr]
        if not cur or cur.char ~= char or not char.Parent then return end
        local m = tpl:Clone()
        for _, o in ipairs(m:GetDescendants()) do
            if o:IsA("Script") or o:IsA("LocalScript") or o:IsA("Humanoid") then o:Destroy() end
        end
        for _, p in ipairs(m:GetDescendants()) do
            if p:IsA("BasePart") then
                p.Anchored = true
                p.CanCollide = false
                p.CanQuery = false
                p.CanTouch = false
                p.CastShadow = false
            end
        end
        m.Parent = workspace
        cur.model = m
    end)
end

local function updateCharModels()
    local id = tonumber((CFG.CharId or ""):match("%d+"))
    if CFG.CharSelf and id and LP.Character then
        ensureCharModel(LP, LP.Character, id)
        setSelfHidden(LP.Character, true)
        selfHidden = LP.Character
    elseif selfHidden then
        setSelfHidden(selfHidden, false)
        selfHidden = nil
        dropCharModel(LP)
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP then
            local c = plr.Character
            if CFG.CharOthers and id and c then
                ensureCharModel(plr, c, id)
            else
                dropCharModel(plr)
            end
        end
    end
    for _, rec in pairs(charRecs) do
        if rec.model and rec.char and rec.char.Parent then
            local hrp = rec.char:FindFirstChild("HumanoidRootPart")
            if hrp then
                if rec.model:IsA("Model") then
                    rec.model:PivotTo(hrp.CFrame)
                else
                    rec.model.CFrame = hrp.CFrame
                end
            end
        end
    end
end

-- ===== bullet tracer =====
local function drawBullet(from, to)
    local d = to - from
    local len = d.Magnitude
    if len < 1 then return end
    local p = Instance.new("Part")
    p.Anchored = true
    p.CanCollide = false
    p.CanQuery = false
    p.CanTouch = false
    p.CastShadow = false
    p.Material = Enum.Material.Neon
    p.Color = rgb(CFG.Col.Beam)
    p.Size = Vector3.new(0.12, 0.12, len)
    p.CFrame = CFrame.lookAt(from, to) * CFrame.new(0, 0, -len / 2)
    p.Parent = workspace
    TweenService:Create(p, TweenInfo.new(CFG.BulletFade, Enum.EasingStyle.Linear), {
        Transparency = 1, Size = Vector3.new(0.02, 0.02, len),
    }):Play()
    game:GetService("Debris"):AddItem(p, CFG.BulletFade + 0.1)
end

        getFx("BlurEffect", {Size = 0}).Size = math.clamp(speed * 0.08, 0, 14)

local function updateFx()
    local char = LP.Character
    if char ~= S.fxChar then buildDeco(char) end
    local d = S.deco
    local root = d.root
    if not (d.folder and root and root.Parent) then
        updateLighting(0); return
    end

    if not (CFG.Aura or CFG.Halo or CFG.Wings or CFG.Trail or CFG.SpinRing or CFG.JumpCircle) then
        for _, s in ipairs({-1, 1}) do d.trails[s].Enabled = false end
        d.em.Rate = 0
        updateLighting(0); return
    end

    local t  = os.clock()
    local cf = root.CFrame

    d.aura.CFrame = cf
    d.aura.Transparency = CFG.Aura and (0.82 + math.sin(t * 2.5) * 0.05) or 1
    d.emPart.CFrame = cf
    d.em.Rate = CFG.Aura and 16 or 0

    local hc = cf * CFrame.new(0, 4.4 + math.sin(t * 2) * 0.12, 0)
    for i, p in ipairs(d.halo) do
        local a = (i / HALO_N) * math.pi * 2 + t * 1.6
        p.CFrame = hc * CFrame.new(math.cos(a) * 1.25, 0, math.sin(a) * 1.25)
        p.Transparency = CFG.Halo and 0.1 or 1
    end

    for i, p in ipairs(d.ring) do
        local a = (i / RING_N) * math.pi * 2 + t * 3
        p.CFrame = cf * CFrame.new(math.cos(a) * 2.2, 0.4, math.sin(a) * 2.2)
            * CFrame.Angles(0, a + math.pi / 2, 0)
        p.Transparency = CFG.SpinRing and 0.15 or 1
    end

    local flap = math.sin(t * 6)
    for _, s in ipairs({-1, 1}) do
        local shoulder = cf * CFrame.new(s * 0.7, 0.8, 0.45)
        for i, p in ipairs(d.wings[s]) do
            local k    = (i - 1) / (FEATHERS - 1)
            local len  = 2.8 - k * 1.6
            local yaw  = -s * (0.25 + k * 0.9)
            local wave = math.sin(t * 6 - k * 2.2) * 0.25
            local roll = s * (0.15 + flap * 0.12 + wave)
            local rot  = shoulder * CFrame.Angles(roll, yaw, 0)
            p.Size = Vector3.new(len, 0.08, 0.5 - k * 0.2)
            p.CFrame = rot * CFrame.new(s * len / 2, 0, 0)
            p.Transparency = CFG.Wings and (0.1 + k * 0.25) or 1
            if i == FEATHERS then d.tips[s].CFrame = rot * CFrame.new(s * len, 0, 0) end
        end
        d.trails[s].Enabled = CFG.Trail
    end

    updateLighting(0)
end

        end
    end

-- ===== anti fling =====
local flingOrig = setmetatable({}, {__mode = "k"})
local ghostedChar = setmetatable({}, {__mode = "k"})

local function ghostPlayer(plr, on)
    local char = plr.Character
    if not char then return end
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then
            if on then
                if flingOrig[p] == nil then flingOrig[p] = p.CanCollide end
                p.CanCollide = false
            elseif flingOrig[p] ~= nil then
                p.CanCollide = flingOrig[p]
                flingOrig[p] = nil
            end
        end
    end
end

S.unghost = function(plr)
    ghostPlayer(plr, false)
    ghostedChar[plr] = nil
end

local function restoreAllGhosts()
    for p, orig in pairs(flingOrig) do
        if p.Parent then p.CanCollide = orig end
        flingOrig[p] = nil
    end
end

local function stabilizeSelf(char)
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") and not p.Anchored then
            if p.AssemblyLinearVelocity.Magnitude > CFG.AntiFlingCap then
                p.AssemblyLinearVelocity = Vector3.zero
            end
            if p.AssemblyAngularVelocity.Magnitude > CFG.AntiFlingCap then
                p.AssemblyAngularVelocity = Vector3.zero
            end
        end
    end
end

RunService.Stepped:Connect(function()
    if not S.alive then return end
    if os.clock() - S.lastFling < 0.15 then return end
    S.lastFling = os.clock()
    if CFG.AntiFling then
        local char = LP.Character
        if char and not S.flinging then stabilizeSelf(char) end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character and ghostedChar[plr] ~= plr.Character and not S.flinging then
                ghostPlayer(plr, true)
                ghostedChar[plr] = plr.Character
            end
        end
    elseif next(flingOrig) then
        restoreAllGhosts()
    end
end)

local function grabGun()
    local drop = workspace:FindFirstChild("GunDrop", true)
    local char = LP.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not (drop and drop:IsA("BasePart") and root) then return end
    if char:FindFirstChild("Gun") or LP.Backpack:FindFirstChild("Gun") then return end
    if firetouchinterest then
        firetouchinterest(root, drop, 0)
        firetouchinterest(root, drop, 1)
    else
        local old = root.CFrame
        root.CFrame = drop.CFrame + Vector3.new(0, 3, 0)
        task.wait(0.15)
        root.CFrame = old
    end
end

task.spawn(function()
    while S.alive and task.wait(1.5) do
        if CFG.AutoGun then pcall(grabGun) end
    end
end)

        end

end)

local function checkbox(parent, label, key, cb)
    local row = new("TextButton", parent, {
        Size = UDim2.new(1, 0, 0, 18), BackgroundTransparency = 1, Text = "", AutoButtonColor = false,
        LayoutOrder = nextOrder(),
    })
    reg(row, label)
    local box = new("Frame", row, {
        Position = UDim2.fromOffset(0, 3), Size = UDim2.fromOffset(12, 12),
        BackgroundColor3 = C.off, BorderSizePixel = 0,
    })
    round(box, 2); stroke(box, C.line)
    new("TextLabel", row, {
        Position = UDim2.fromOffset(20, 0), Size = UDim2.new(1, -90, 1, 0),
        BackgroundTransparency = 1, Text = label, Font = FONT, TextSize = 12,
        TextColor3 = C.text, TextXAlignment = Enum.TextXAlignment.Left,
    })
    local opt = new("TextButton", row, {
        Position = UDim2.new(1, -66, 0, 0), Size = UDim2.fromOffset(66, 18),
        BackgroundTransparency = 1, Text = "...", Font = FONTB, TextSize = 12,
        TextColor3 = C.dim, TextXAlignment = Enum.TextXAlignment.Right, AutoButtonColor = false,
    })

    local lastOn
    local function paint()
        local on = CFG[key] == true
        if on ~= lastOn then
            lastOn = on
            tween(box, 0.15, {BackgroundColor3 = on and C.on or C.off})
        end
        if S.listening == key then
            opt.Text, opt.TextColor3 = "...", C.on
        else
            opt.Text, opt.TextColor3 = BINDS[key] or "...", C.dim
        end
    end
    local function flip()
        CFG[key] = not CFG[key]
        if cb then cb(CFG[key]) end
        refreshAll()
    end

    row.MouseButton1Click:Connect(flip)
    row.MouseButton2Click:Connect(function() armBind(key, label) end)
    opt.MouseButton1Click:Connect(function() armBind(key, label) end)
    BINDACT[key] = flip
    addBindRow(key, label, function() return CFG[key] == true end)
    table.insert(refreshers, paint)
    paint()
end

local function slider(parent, label, key, min, max, step, fmt)
    local row = new("Frame", parent, {Size = UDim2.new(1, 0, 0, 30), BackgroundTransparency = 1, LayoutOrder = nextOrder()})
    reg(row, label)
    new("TextLabel", row, {
        Size = UDim2.new(1, -80, 0, 14), BackgroundTransparency = 1, Text = label,
        Font = FONT, TextSize = 12, TextColor3 = C.text, TextXAlignment = Enum.TextXAlignment.Left,
    })
    local val = new("TextLabel", row, {
        Position = UDim2.new(1, -80, 0, 0), Size = UDim2.new(0, 80, 0, 14), BackgroundTransparency = 1,
        Font = FONT, TextSize = 12, TextColor3 = C.dim, TextXAlignment = Enum.TextXAlignment.Right,
    })
    local bar = new("Frame", row, {
        Position = UDim2.fromOffset(0, 20), Size = UDim2.new(1, 0, 0, 3),
        BackgroundColor3 = C.off, BorderSizePixel = 0,
    })
    local fill = new("Frame", bar, {Size = UDim2.new(0, 0, 1, 0), BackgroundColor3 = C.on, BorderSizePixel = 0})
    local hit = new("TextButton", bar, {
        Size = UDim2.new(1, 0, 0, 14), Position = UDim2.fromOffset(0, -6),
        BackgroundTransparency = 1, Text = "", AutoButtonColor = false,
    })
    local function paint()
        local v = CFG[key]
        fill.Size = UDim2.new((v - min) / (max - min), 0, 1, 0)
        val.Text = fmt and fmt(v) or tostring(v)
    end
    local function setX(x)
        local pct = math.clamp((x - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
        CFG[key] = math.clamp(min + math.floor((max - min) * pct / step + 0.5) * step, min, max)
        paint()
    end
    local drag = false
    hit.InputBegan:Connect(function(i)
        if isPress(i) then drag = true; setX(i.Position.X) end
    end)
    UIS.InputChanged:Connect(function(i)
        if drag and isMove(i) then setX(i.Position.X) end
    end)
    UIS.InputEnded:Connect(function(i)
        if isPress(i) then drag = false end
    end)
    paint()
    table.insert(refreshers, paint)
end

-- ===== teleport helpers =====
local function tpTo(cf)
    local c = LP.Character
    local r = c and c:FindFirstChild("HumanoidRootPart")
    if r then r.CFrame = cf end
end

-- the map is the biggest model in workspace that is not a player, a humanoid or our own folders
local function findMapModel()
    local best, bestN = nil, 0
    for _, m in ipairs(workspace:GetChildren()) do
        if m:IsA("Model") and not Players:GetPlayerFromCharacter(m)
            and not m:FindFirstChildOfClass("Humanoid") and not string.find(m.Name, "nlz", 1, true) then
            local n = #m:GetDescendants()
            if n > bestN then best, bestN = m, n end
        end
    end
    return best
end

local function tpToMap()
    local m = findMapModel()
    if not m then toast("map not found") return end
    local cf, size = m:GetBoundingBox()
    tpTo(CFrame.new(cf.Position + Vector3.new(0, size.Y / 2 + 5, 0)))
end

local function tpToGunDrop()
    local drop = workspace:FindFirstChild("GunDrop", true)
    if not (drop and drop:IsA("BasePart")) then toast("no gun drop on the map") return end
    tpTo(drop.CFrame + Vector3.new(0, 4, 0))
end

local function tpToPlayer(plr)
    local tr = plr and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
    if not tr then toast("no target (set fling mode or name)") return end
    tpTo(tr.CFrame + Vector3.new(0, 3, 0))
end

-- ===== griddy emote =====
-- plays the emote as an action track on the humanoid's animator. it does not lock movement,
-- so running and jumping keep working while it loops.
local griddyTrack, griddyChar = nil, nil
local function updateGriddy()
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not (CFG.GriddyOn and hum and hum.Health > 0) then
        if griddyTrack then griddyTrack:Stop(0.2); griddyTrack = nil end
        return
    end
    if griddyTrack and griddyChar == char then return end
    local id = nil
    for digits in string.gmatch(CFG.GriddyId or "", "%d+") do
        if not id or #digits > #id then id = digits end
    end
    id = id and tonumber(id)
    if not id then
        CFG.GriddyOn = false
        toast("set griddy anim id first")
        return
    end
    if griddyTrack then griddyTrack:Stop(0.2) end
    local animator = hum:FindFirstChildOfClass("Animator") or Instance.new("Animator", hum)
    local anim = Instance.new("Animation")
    anim.AnimationId = "rbxassetid://" .. id
    local ok, res = pcall(function()
        local t = animator:LoadAnimation(anim)
        t.Priority = Enum.AnimationPriority.Action
        t.Looped = true
        t:Play(0.2)
        return t
    end)
    if not ok then
        print("[nlz griddy] load/play failed:", tostring(res))
        toast("griddy failed: " .. tostring(res):sub(1, 60))
        CFG.GriddyOn = false
        return
    end
    print("[nlz griddy] playing id " .. tostring(id) .. ", length " .. tostring(res.Length))
    griddyTrack = res
    griddyChar = char
end

S.griddyUpdate = updateGriddy

-- ===== config / unload =====
function saveConfig()
    if not writefile then toast("writefile not supported"); return end
    pcall(function() writefile(CONFIG_FILE, HttpService:JSONEncode({cfg = CFG, binds = BINDS})) end)
    toast("config saved")
end
function loadConfig()
    if not (readfile and isfile and isfile(CONFIG_FILE)) then toast("no config found"); return end
    local ok, data = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
    if not ok or type(data) ~= "table" then toast("config corrupt"); return end
    for k, v in pairs(data.cfg or {}) do
        if CFG[k] ~= nil and typeof(v) == typeof(CFG[k]) then CFG[k] = v end
    end
    for k in pairs(BINDS) do BINDS[k] = nil end
    for k, v in pairs(data.binds or {}) do BINDS[k] = v end
    refreshAll()
    toast("config loaded")
end
function resetConfig()
    for k, v in pairs(DEFAULTS) do CFG[k] = v end
    refreshAll()
    toast("defaults restored")
end
function unload()
    S.alive = false
    for plr in pairs(esp) do removeEsp(plr) end
    for plr, h in pairs(chams) do h:Destroy(); chams[plr] = nil end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Character then setSkin(plr.Character, false) end
    end
    restoreAllGhosts()
    clearDeco()
    if S.hat then S.hat:Destroy(); S.hat = nil end
    clearAura()
    clearGun()
    killProcAura()
    killProcTrail()
    killFly()
    if selfHidden then setSelfHidden(selfHidden, false) end
    for plr in pairs(charRecs) do dropCharModel(plr) end
    if S.wx then S.wx.part:Destroy(); S.wx = nil end
    for _, e in pairs(S.fxLight) do e:Destroy() end
    S.fxLight = {}
    local L = S.lightOrig
    Lighting.FogStart, Lighting.FogEnd, Lighting.FogColor = L.FogStart, L.FogEnd, L.FogColor
    fovC:Remove()
    if S.espClear then S.espClear() end
    gui:Destroy()
end

-- ===== player esp, gui based =====
do
    local store = {}

    local function seg(f, px, py, sx, sy)
        f.Position = UDim2.fromOffset(px, py)
        f.Size = UDim2.fromOffset(sx, sy)
    end

    local function build(plr)
        local root = new("Frame", gui, {
            Name = "nlz_esp_" .. plr.Name, Position = UDim2.fromOffset(0, 0),
            Size = UDim2.fromOffset(0, 0), BackgroundTransparency = 1, Visible = false, ZIndex = 2,
        })
        local t = {root = root}
        t.fill = new("Frame", root, {BackgroundColor3 = C.white, BackgroundTransparency = 0.88, BorderSizePixel = 0, ZIndex = 2})
        new("UIGradient", t.fill, {Rotation = 90, Color = ColorSequence.new(C.white, C.dim)})
        t.box = new("Frame", root, {BackgroundTransparency = 1, BorderSizePixel = 0, ZIndex = 3})
        t.boxStroke = new("UIStroke", t.box, {Color = C.white, Thickness = 1, ApplyStrokeMode = Enum.ApplyStrokeMode.Border})
        t.shadow, t.corner = {}, {}
        for i = 1, 8 do
            t.shadow[i] = new("Frame", root, {BackgroundColor3 = C.pill, BackgroundTransparency = 0.35, BorderSizePixel = 0, ZIndex = 3})
            t.corner[i] = new("Frame", root, {BackgroundColor3 = C.white, BorderSizePixel = 0, ZIndex = 4})
        end
        t.hpBack = new("Frame", root, {BackgroundColor3 = C.pill, BorderSizePixel = 0, ZIndex = 3})
        t.hp = new("Frame", t.hpBack, {
            AnchorPoint = Vector2.new(0, 1), Position = UDim2.fromScale(0, 1), Size = UDim2.fromScale(1, 1),
            BackgroundColor3 = C.inno, BorderSizePixel = 0, ZIndex = 4,
        })
        t.pill = new("Frame", root, {BackgroundColor3 = C.pill, BackgroundTransparency = 0.15, BorderSizePixel = 0, ZIndex = 4})
        round(t.pill, 4)
        t.name = new("TextLabel", t.pill, {
            Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Font = Enum.Font.Code,
            TextSize = 13, TextColor3 = C.text, ZIndex = 5, Text = "",
        })
        t.role = new("TextLabel", root, {
            Size = UDim2.fromOffset(120, 14), BackgroundTransparency = 1, Font = FONTB,
            TextSize = 11, ZIndex = 4, Text = "",
        })
        t.dist = new("TextLabel", root, {
            Size = UDim2.fromOffset(120, 14), BackgroundTransparency = 1, Font = Enum.Font.Code,
            TextSize = 12, TextColor3 = C.white, ZIndex = 4, Text = "",
        })
        t.glow1 = Drawing.new("Line"); t.glow1.Thickness = 6; t.glow1.Visible = false
        t.glow2 = Drawing.new("Line"); t.glow2.Thickness = 3; t.glow2.Visible = false
        t.line  = Drawing.new("Line"); t.line.Thickness = 1;  t.line.Visible = false
        store[plr] = t
        return t
    end

    S.espUpdate = function(plr, cam, col)
        local t = store[plr] or build(plr)
        t.line.Visible = false
        t.glow1.Visible = false
        t.glow2.Visible = false
        local char = plr.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        local head = char and char:FindFirstChild("Head")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if not (CFG.ESP and hrp and head and hum and hum.Health > 0) then
            t.root.Visible = false; return
        end

        local dist = (cam.CFrame.Position - hrp.Position).Magnitude
        if dist > CFG.MaxDist then t.root.Visible = false; return end
        local alpha = CFG.FadeDist and math.clamp(1 - dist / CFG.MaxDist * 0.6, 0.4, 1) or 1

        local topP, topOn = cam:WorldToViewportPoint(head.Position + Vector3.new(0, 0.7, 0))
        local botP, botOn = cam:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3.1, 0))
        if not (topOn and botOn) then t.root.Visible = false; return end
        local h = botP.Y - topP.Y
        if h < 4 then t.root.Visible = false; return end
        t.root.Visible = true

        local w      = h * 0.55
        local cx     = cam:WorldToViewportPoint(hrp.Position).X
        local x, y   = cx - w / 2, topP.Y
        local bottom = y + h
        local L      = math.clamp(math.min(w, h) * 0.24, 5, 30)

        t.fill.Visible = CFG.Box
        seg(t.fill, x, y, w, h)
        t.fill.BackgroundTransparency = 0.88 + (1 - alpha) * 0.12

        t.box.Visible = CFG.Box and not CFG.CornerBox
        seg(t.box, x, y, w, h)
        t.boxStroke.Color = col
        t.boxStroke.Transparency = 1 - alpha

        local segs = {
            {x, y, L, 2}, {x, y, 2, L},
            {x + w - L, y, L, 2}, {x + w - 2, y, 2, L},
            {x, bottom - 2, L, 2}, {x, bottom - L, 2, L},
            {x + w - L, bottom - 2, L, 2}, {x + w - 2, bottom - L, 2, L},
        }
        for i = 1, 8 do
            local s  = segs[i]
            local c, sh = t.corner[i], t.shadow[i]
            seg(sh, s[1] - 1, s[2] - 1, s[3] + 2, s[4] + 2)
            seg(c, s[1], s[2], s[3], s[4])
            c.BackgroundColor3 = col
            c.BackgroundTransparency = 1 - alpha
            sh.BackgroundTransparency = 1 - alpha * 0.65
            c.Visible = CFG.CornerBox
            sh.Visible = CFG.CornerBox
        end

        local pillW = #plr.Name * 7 + 18
        t.pill.Visible = CFG.Name
        seg(t.pill, cx - pillW / 2, y - 24, pillW, 16)
        t.pill.BackgroundTransparency = 1 - alpha * 0.85
        t.name.Text = plr.Name
        t.name.TextTransparency = 1 - alpha

        t.role.Visible = CFG.Role
        t.role.Text = string.upper(roleOf(plr))
        t.role.TextColor3 = col
        t.role.TextTransparency = 1 - alpha
        t.role.Position = UDim2.fromOffset(cx - 60, y - 40)

        t.dist.Visible = CFG.Distance
        t.dist.Text = math.floor(dist) .. "m"
        t.dist.TextTransparency = 1 - alpha
        t.dist.Position = UDim2.fromOffset(cx - 60, bottom + 3)

        local pct = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
        t.hpBack.Visible = CFG.HealthBar
        seg(t.hpBack, x - 6, y - 1, 3, h + 2)
        t.hpBack.BackgroundTransparency = 1 - alpha
        t.hp.Size = UDim2.fromScale(1, pct)
        t.hp.BackgroundColor3 = Color3.fromHSV(pct * 0.33, 1, 1)
        t.hp.BackgroundTransparency = 1 - alpha

        local rp, rpOn = cam:WorldToViewportPoint(hrp.Position)
        local vp   = cam.ViewportSize
        local from = CFG.TracerFrom == "center" and Vector2.new(vp.X / 2, vp.Y / 2) or Vector2.new(vp.X / 2, vp.Y)
        local tc   = rgb(CFG.Col.Tracer)
        local show = CFG.Tracer and rpOn
        local to   = Vector2.new(rp.X, rp.Y)
        t.glow1.From, t.glow1.To = from, to
        t.glow1.Color, t.glow1.Transparency = tc, alpha * 0.22
        t.glow1.Visible = show
        t.glow2.From, t.glow2.To = from, to
        t.glow2.Color, t.glow2.Transparency = tc, alpha * 0.45
        t.glow2.Visible = show
        t.line.From, t.line.To = from, to
        t.line.Color, t.line.Transparency = tc, alpha
        t.line.Visible = show
    end

    S.espClear = function()
        for p, t in pairs(store) do
            t.line:Remove(); t.glow1:Remove(); t.glow2:Remove()
            t.root:Destroy()
            store[p] = nil
        end
    end

    Players.PlayerRemoving:Connect(function(p)
        if store[p] then
            store[p].line:Remove(); store[p].glow1:Remove(); store[p].glow2:Remove()
            store[p].root:Destroy()
            store[p] = nil
        end
    end)
end

UIS.InputBegan:Connect(function(i, gp)
    if not S.alive then return end

    if S.listening and i.UserInputType == Enum.UserInputType.Keyboard then
        if i.KeyCode == Enum.KeyCode.Backspace or i.KeyCode == Enum.KeyCode.Escape then
            BINDS[S.listening] = nil
        else
            BINDS[S.listening] = i.KeyCode.Name
        end
        S.listening = nil
        refreshAll()
        return
    end

    if i.UserInputType == Enum.UserInputType.MouseButton2 and not gp then
        S.aimHeld = true
    end

    if i.UserInputType == Enum.UserInputType.Keyboard and i.KeyCode == Enum.KeyCode.G then
        main.Visible = not main.Visible
        return
    end

    if gp then return end
    if i.UserInputType == Enum.UserInputType.Keyboard then
        if i.KeyCode == Enum.KeyCode[CFG.SilentKey] then
            silentFire()
        end
        if i.KeyCode == Enum.KeyCode[CFG.WbKey] then
            wbShoot()
        end
        for id, kc in pairs(BINDS) do
            if i.KeyCode.Name == kc and BINDACT[id] then
                BINDACT[id]()
            end
        end
    end
end)

UIS.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton2 then S.aimHeld = false end
end)
